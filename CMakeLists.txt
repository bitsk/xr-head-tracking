cmake_minimum_required(VERSION 3.16)
project(xr_head_tracking LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(XRDevice CONFIG REQUIRED)
include(GNUInstallDirs)
find_package(XRHexagonRPC CONFIG REQUIRED)
foreach(kind executable plugin)
  if(kind STREQUAL executable)
    add_executable(a081-qvr-stock-live-stream src/main.cpp src/qvr_bridge.cpp)
    set(target a081-qvr-stock-live-stream)
  else()
    add_library(a081-qvr-client-live-stream SHARED src/main.cpp src/qvr_bridge.cpp)
    set(target a081-qvr-client-live-stream)
    target_compile_definitions(${target} PRIVATE A081_QVR_PLUGIN)
  endif()
  target_include_directories(${target} PRIVATE include src)
  target_compile_options(${target} PRIVATE -O2 -Wall -Wextra)
  target_link_libraries(${target} PRIVATE XRDevice::abi XRHexagonRPC::client pthread dl)
endforeach()
add_executable(a081-tracking-observe src/tracking-observe.c)
target_include_directories(a081-tracking-observe PRIVATE include)

target_compile_definitions(a081-tracking-observe PRIVATE _GNU_SOURCE)

install(TARGETS a081-qvr-stock-live-stream RUNTIME DESTINATION ${CMAKE_INSTALL_LIBDIR}/xr/head-tracking)
install(TARGETS a081-qvr-client-live-stream LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/xr/head-tracking)
set_target_properties(a081-tracking-observe PROPERTIES OUTPUT_NAME xr-head-observe)
install(TARGETS a081-tracking-observe RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/xr DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})


install(PROGRAMS tools/audit-motion.py DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME xr-head-audit-motion)
